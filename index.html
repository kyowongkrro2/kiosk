/** * 학습개발2팀 키오스크 — 저장 서버 (Google Apps Script) * * 동시 저장 충돌 방지: * - 데이터에 version 번호를 붙여 저장합니다. * - 브라우저는 "내가 마지막으로 본 version"과 함께 저장을 요청합니다. * - 그 사이 다른 사람이 먼저 저장했다면 거절하고 최신 데이터를 돌려줍니다. * → 브라우저가 자기 변경을 최신 데이터 위에 다시 얹어 재저장하므로 아무것도 사라지지 않습니다. * - 확인→저장 과정은 LockService로 한 번에 한 명씩만 처리됩니다. * * 데이터는 내 Google 드라이브의 "kiosk-data.json" 파일에 저장됩니다. */ const DATA_FILE_NAME = "kiosk-data.json"; /** 처음 한 번 편집기에서 이 함수를 선택해 ▶실행 → 권한 허용 */ function setup() { const f = getFile_(); Logger.log("데이터 파일 준비 완료: " + f.getUrl()); } function doGet(e) { return json_(readState_()); } function doPost(e) { const lock = LockService.getScriptLock(); if (!lock.tryLock(20000)) { return json_({ ok: false, error: "busy" }); } try { const body = JSON.parse(e.postData.contents); const props = PropertiesService.getScriptProperties(); const current = Number(props.getProperty("DATA_VERSION") || 0); if (Number(body.baseVersion) !== current) { const st = readState_(); return json_({ ok: false, conflict: true, version: st.version, data: st.data }); } getFile_().setContent(JSON.stringify(body.data)); const next = current + 1; props.setProperty("DATA_VERSION", String(next)); return json_({ ok: true, version: next }); } catch (err) { return json_({ ok: false, error: String(err) }); } finally { lock.releaseLock(); } } /* ── 내부 함수 ───────────────────────────────────────── */ function readState_() { const props = PropertiesService.getScriptProperties(); const version = Number(props.getProperty("DATA_VERSION") || 0); const text = getFile_().getBlob().getDataAsString("UTF-8"); let data = null; try { data = text ? JSON.parse(text) : null; } catch (err) { data = null; } return { version: version, data: data }; } function getFile_() { const props = PropertiesService.getScriptProperties(); const id = props.getProperty("DATA_FILE_ID"); if (id) { try { return DriveApp.getFileById(id); } catch (err) { /* 파일이 지워졌으면 새로 만듦 */ } } const f = DriveApp.createFile(DATA_FILE_NAME, "null", MimeType.PLAIN_TEXT); props.setProperty("DATA_FILE_ID", f.getId()); props.setProperty("DATA_VERSION", "0"); return f; } function json_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
